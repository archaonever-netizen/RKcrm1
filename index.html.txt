<body><div id="app"></div>
<script>
const API = window.location.origin + '/api';
// Пример авторизации
async function login() {
  const user = await (await fetch(API+'/login',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({username: document.getElementById('username').value})})).json();
  if (user.user) {
    document.getElementById('app').innerHTML = `<h2>Привет, ${user.user.firstName}</h2><p>Роль: ${user.user.role}</p>
    <button onclick="loadProposals()">Акции / Скидки</button>`;
    window.user = user.user;
  } else alert('Ошибка входа');
}
async function loadProposals() {
  const resp = await fetch(API+'/proposals');
  const list = await resp.json();
  document.getElementById('app').innerHTML += list.map(p=>`<div class="card">${p.developerName}<br>${p.text}</div>`).join('');
}
// Инициализация
document.getElementById('app').innerHTML = '<input id="username" placeholder="Логин"><button onclick="login()">Войти</button>';
</script></body></html>
